# Testboard i18n (English / Ukrainian) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every viewer can use the Testboard UI in English or Ukrainian (browser default, manual override), while Slack, delivery-log and AI texts become English for everyone.

**Architecture:** `next-intl` 3.x without i18n routing: the locale is resolved per request from a `NEXT_LOCALE` cookie or `Accept-Language`, messages live in `messages/{en,uk}.json` (ICU), and a server action flips the cookie. Pure helpers (`i18n/locale.ts`, `i18n/dates.ts`) carry the logic and are unit-tested with Jest (`next/jest`); message completeness and leftover hard-coded strings are enforced by Jest tests too.

**Tech Stack:** Next 14.2 App Router, React 18, next-intl 3.26.x, Jest 29 via `next/jest`, `intl-messageformat`.

**Spec:** `docs/superpowers/specs/2026-10-14-i18n-en-uk-design.md`

## Global Constraints

- Supported locales: exactly `en` and `uk`; `en` is the fallback.
- Cookie: `NEXT_LOCALE`, 1 year, `path=/`, `SameSite=Lax`.
- Default: first `Accept-Language` preference starting with `uk` → `uk`, anything else → `en`.
- No URL prefixes; URLs stay `/reports`, `/settings`, …
- `messages/en.json` is the source of truth; `uk.json` must have identical keys and placeholders.
- ICU plurals; Ukrainian plural messages must define `one`, `few`, `many`, `other`.
- Dates follow the UI language: `en` → `Intl` locale `en-GB`, `uk` → `uk`. Dates are formatted on the client.
- Not translated: test titles, errors/stack traces, CI metadata values, project/tag names, the Playwright HTML report.
- Slack messages, delivery-log `detail` strings and AI prompts become English for everyone.
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. Messy `Accept-Language` headers (`uk-UA,uk;q=0.9`, `ru,uk;q=0.8`, `*`, empty, missing) must resolve predictably (uk only when Ukrainian is the top preference) — Task 1 tests.
2. A tampered or stale cookie (`de`, `UK`, `../x`, empty) must be ignored, never crash or load a non-existent messages file; `setLocale` must reject it — Tasks 1 and 3 tests.
3. Ukrainian plural boundaries (0, 1, 2, 4, 5, 11, 12, 21, 22, 25, 111) must pick the right form — Task 2 test.
4. A Ukrainian message missing or renaming a placeholder (`{count}` vs `{n}`) must fail the check, not render a raw `{count}` — Task 2 test.
5. "Today/Yesterday" around midnight and across New Year, and the year suffix only for other years — Task 4 tests with a fixed `now`.

```ts
// i18n/locale.ts
export const LOCALES = ['en', 'uk'] as const;
export type Locale = (typeof LOCALES)[number];
export const DEFAULT_LOCALE: Locale = 'en';
export const LOCALE_COOKIE = 'NEXT_LOCALE';

export const isLocale = (value: unknown): value is Locale =>
  typeof value === 'string' && (LOCALES as readonly string[]).includes(value);

// Highest-weighted Accept-Language entry; ties keep header order.
const topLanguage = (header: string): string | undefined =>
  header
    .split(',')
    .map((part, index) => {
      const [tag, ...params] = part.trim().split(';');
      const q = params.map((p) => p.trim()).find((p) => p.startsWith('q='));

      return { tag: tag.trim().toLowerCase(), q: q ? Number(q.slice(2)) || 0 : 1, index };
    })
    .filter((entry) => entry.tag && entry.q > 0)
    .sort((a, b) => b.q - a.q || a.index - b.index)[0]?.tag;

// Cookie (an explicit choice) wins; otherwise Ukrainian only when it's the
// viewer's top browser preference.
export const resolveLocale = (cookie: string | undefined, acceptLanguage: string | null | undefined): Locale => {
  if (isLocale(cookie)) return cookie;
  const top = acceptLanguage ? topLanguage(acceptLanguage) : undefined;

  return top?.startsWith('uk') ? 'uk' : DEFAULT_LOCALE;
};

// Intl locale used for dates: en-GB gives "8 Jul" and 24h time.
export const intlLocale = (locale: Locale): string => (locale === 'uk' ? 'uk' : 'en-GB');
```

```ts
// i18n/messages.test.ts
import { IntlMessageFormat } from 'intl-messageformat';
import { parse, TYPE, type MessageFormatElement } from '@formatjs/icu-messageformat-parser';

import en from '@/messages/en.json';
import uk from '@/messages/uk.json';

type Tree = { [key: string]: string | Tree };

const flatten = (tree: Tree, prefix = ''): Record<string, string> =>
  Object.entries(tree).reduce<Record<string, string>>((acc, [key, value]) => {
    const path = prefix ? `${prefix}.${key}` : key;

    return typeof value === 'string' ? { ...acc, [path]: value } : { ...acc, ...flatten(value, path) };
  }, {});

const placeholders = (elements: MessageFormatElement[], out = new Set<string>()): Set<string> => {
  for (const el of elements) {
    if ('value' in el && el.type !== TYPE.literal && el.type !== TYPE.pound) out.add(String(el.value));
    if (el.type === TYPE.plural || el.type === TYPE.select) {
      Object.values(el.options).forEach((o) => placeholders(o.value, out));
    }
    if (el.type === TYPE.tag) placeholders(el.children, out);
  }

  return out;
};

const pluralOptionSets = (elements: MessageFormatElement[]): string[][] =>
  elements.flatMap((el) =>
    el.type === TYPE.plural
      ? [Object.keys(el.options), ...Object.values(el.options).flatMap((o) => pluralOptionSets(o.value))]
      : [],
  );

const enFlat = flatten(en as Tree);
const ukFlat = flatten(uk as Tree);

describe('messages', () => {
  it('en and uk have the same keys', () => {
    expect(Object.keys(ukFlat).sort()).toEqual(Object.keys(enFlat).sort());
  });

  it.each(Object.keys(enFlat))('%s parses and has matching placeholders', (key) => {
    const enPh = [...placeholders(parse(enFlat[key]))].sort();
    const ukPh = [...placeholders(parse(ukFlat[key] ?? ''))].sort();

    expect(ukPh).toEqual(enPh);
  });

  it('Ukrainian plurals define one/few/many/other', () => {
    for (const [key, message] of Object.entries(ukFlat)) {
      for (const options of pluralOptionSets(parse(message))) {
        const categories = options.filter((o) => !o.startsWith('='));

        expect({ key, categories: categories.sort() }).toEqual({ key, categories: ['few', 'many', 'one', 'other'] });
      }
    }
  });

  it.each([
    [0, '0 прогонів'],
    [1, '1 прогін'],
    [2, '2 прогони'],
    [4, '4 прогони'],
    [5, '5 прогонів'],
    [11, '11 прогонів'],
    [12, '12 прогонів'],
    [21, '21 прогін'],
    [22, '22 прогони'],
    [25, '25 прогонів'],
    [111, '111 прогонів'],
```

```ts
const ALLOWED = /^(Testboard|Playwright|Slack|Google|Drone|Cron|Haiku|Sonnet|Opus|API|URL|JSON|CI|ZIP|PDF|HTML|EN|UK|env|cron|[0-9%.:·—–\-\s]+)$/;

const ATTR = /\b(?:label|title|placeholder|description|aria-label|subtitle|alt)=(["'])([^"'{}]*[A-Za-zА-Яа-я][^"'{}]*)\1/g;
const TOAST = /toast\.(?:success|error|warning|info)\(\s*(['`])([^'`]*[A-Za-zА-Яа-я][^'`]*)\1/g;

const findLiterals = (source: string): string[] => {
  const lines = source.split('\n').filter((l) => !l.includes('i18n-ignore') && !l.trim().startsWith('//'));
  const text = lines.join('\n');
  const hits: string[] = [];

  for (const re of [JSX_TEXT, ATTR, TOAST]) {
    for (const match of text.matchAll(re)) {
      const value = (match[2] ?? match[1]).trim();

      if (value && !ALLOWED.test(value)) hits.push(value);
    }
  }

  return hits;
};

describe('no hard-coded UI strings', () => {

    expect(findLiterals(fs.readFileSync(path.join(process.cwd(), file), 'utf8'))).toEqual([]);
```

```ts
// app/lib/locale-actions.ts
'use server';

import { cookies } from 'next/headers';

import { isLocale, LOCALE_COOKIE } from '@/i18n/locale';

const ONE_YEAR = 60 * 60 * 24 * 365;

export async function setLocale(locale: string): Promise<{ ok: boolean }> {
  if (!isLocale(locale)) return { ok: false };
  cookies().set(LOCALE_COOKIE, locale, { maxAge: ONE_YEAR, path: '/', sameSite: 'lax' });

  return { ok: true };
}
```

```ts
// i18n/dates.ts
export type DateLabels = { today: (time: string) => string; yesterday: (time: string) => string };
export type HeadingLabels = { today: (day: string) => string; yesterday: (day: string) => string };
export type AgoLabels = {
  justNow: string;
  minutes: (n: number) => string;
  hours: (n: number) => string;
  days: (n: number) => string;
};

export const dayKey = (date: Date | string): string => new Date(date).toDateString();

const isYesterday = (date: Date, now: Date) => {
  const y = new Date(now);

  y.setDate(now.getDate() - 1);

  return dayKey(date) === dayKey(y);
};

const shortDay = (date: Date, intl: string, now: Date) =>
  date.toLocaleDateString(intl, {
    day: 'numeric',
    month: 'short',
    ...(date.getFullYear() !== now.getFullYear() && { year: 'numeric' }),
  });

// "Today, 18:46" / "Yesterday, 18:46" / "8 Jul, 18:46" / "31 Dec 2025, 09:00"
export const formatRelativeDay = (date: Date, intl: string, labels: DateLabels, now = new Date()): string => {
  const time = date.toLocaleTimeString(intl, { hour: '2-digit', minute: '2-digit' });

  if (dayKey(date) === dayKey(now)) return labels.today(time);
  if (isYesterday(date, now)) return labels.yesterday(time);

  return `${shortDay(date, intl, now)}, ${time}`;
};

// Day separator in lists: "Today · 8 Jul" / "Yesterday · 7 Jul" / "28 Jun"
export const formatDayHeading = (date: Date, intl: string, labels: HeadingLabels, now = new Date()): string => {
  const day = shortDay(date, intl, now);

  if (dayKey(date) === dayKey(now)) return labels.today(day);
  if (isYesterday(date, now)) return labels.yesterday(day);

  return day;
};

// "just now" / "5 min ago" / "3 h ago" / "2 d ago"; future times read as "just now".
export const formatAgo = (from: Date, labels: AgoLabels, now = new Date()): string => {
  const minutes = Math.floor(Math.max(0, now.getTime() - from.getTime()) / 60000);

  if (minutes < 1) return labels.justNow;
  if (minutes < 60) return labels.minutes(minutes);
  const hours = Math.floor(minutes / 60);

  if (hours < 24) return labels.hours(hours);

  return labels.days(Math.floor(hours / 24));
};
```

```tsx
// app/components/date-format.tsx
'use client';
import { useEffect, useState } from 'react';
import { useLocale, useTranslations } from 'next-intl';

import { formatRelativeDay, type AgoLabels, type DateLabels, type HeadingLabels } from '@/i18n/dates';
import { intlLocale, type Locale } from '@/i18n/locale';

export const useDateLabels = () => {
  const t = useTranslations('dates');
  const intl = intlLocale(useLocale() as Locale);
  const relative: DateLabels = { today: (time) => t('today', { time }), yesterday: (time) => t('yesterday', { time }) };
  const heading: HeadingLabels = { today: (day) => t('todayDay', { day }), yesterday: (day) => t('yesterdayDay', { day }) };
  const ago: AgoLabels = {
    justNow: t('justNow'),
    minutes: (count) => t('minutesAgo', { count }),
    hours: (count) => t('hoursAgo', { count }),
    days: (count) => t('daysAgo', { count }),
  };

  return { intl, relative, heading, ago };
};

/** Formats on the client — server and viewer time zones may differ. */
export default function FormattedDate({ date, options, relativeDay }: { date: Date; options?: Intl.DateTimeFormatOptions; relativeDay?: boolean }) {
  const { intl, relative } = useDateLabels();
  const [formatted, setFormatted] = useState('');
  const [full, setFull] = useState<string>();

  useEffect(() => {
    const d = new Date(date);

    setFormatted(relativeDay ? formatRelativeDay(d, intl, relative) : d.toLocaleString(intl, options));
    setFull(options || relativeDay ? d.toLocaleString(intl) : undefined);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [date, relativeDay, intl, JSON.stringify(options)]);

  return <span className="whitespace-nowrap" title={full}>{formatted}</span>;
}
```
